   283→   283→function BelanjaRow({

   405→   405→function EditBelanjaDialog({
   406→   406→  tx,

   408→   408→}: {
   409→   409→  tx: Transaction;
   410→   410→  onClose: () => void;
   411→   411→}) {
   412→   412→  const updateMutation = useUpdateTransaction();

   414→   414→  const { data: vendorsData } = useVendors();
   415→   415→  const vendors = vendorsData?.items ?? [];
   416→   416→
   417→   417→  const [form, setForm] = useState({
   418→   418→    noBku: tx.noBku || "",
   419→   419→    noPesan: tx.noPesan || "",
   420→   420→    tglPesan: tx.tglPesan && tx.tglPesan.includes("-")
   421→   421→      ? tx.tglPesan.substring(0, 10)
   422→   422→      : tx.tglPesan || "",
   423→   423→    tglBast: tx.tglBast && tx.tglBast.includes("-")
   424→   424→      ? tx.tglBast.substring(0, 10)
   425→   425→      : tx.tglBast || "",
   426→   426→    tglBayar: tx.tglBayar && tx.tglBayar.includes("-")
   427→   427→      ? tx.tglBayar.substring(0, 10)
   428→   428→      : tx.tglBayar || "",
   429→   429→    uraian: tx.uraian || "",
   430→   430→    namaBarang: tx.namaBarang || "",
   431→   431→    volume: String(tx.volume || 0),
   432→   432→    satuan: tx.satuan || "",
   433→   433→    tarifHarga: String(tx.tarifHarga || 0),
   434→   434→    vendorId: tx.vendorId || "__none__",
   435→   435→    bulan: String(tx.bulan || ""),
   436→   436→    status: tx.status || "pending",
   437→   437→  });
   438→   438→
   439→   439→  const computedJumlah =
   440→   440→    (parseFloat(form.volume) || 0) * (parseFloat(form.tarifHarga) || 0);
   441→   441→
   442→   442→  const handleSave = async () => {
   443→   443→    try {
   444→   444→      await updateMutation.mutateAsync({
   445→   445→        id: tx.id,
   446→   446→        data: {
   447→   447→          noBku: form.noBku || null,
   448→   448→          noPesan: form.noPesan || null,
   449→   449→          tglPesan: form.tglPesan || null,
   450→   450→          tglBast: form.tglBast || null,
   451→   451→          tglBayar: form.tglBayar || null,
   452→   452→          uraian: form.uraian,
   453→   453→          namaBarang: form.namaBarang || null,
   454→   454→          volume: parseFloat(form.volume) || 0,
   455→   455→          satuan: form.satuan || null,
   456→   456→          tarifHarga: parseFloat(form.tarifHarga) || 0,
   457→   457→          jumlah: computedJumlah,
   458→   458→          realisasi: computedJumlah,
   459→   459→          vendorId: form.vendorId && form.vendorId !== "__none__" ? form.vendorId : null,
   460→   460→          bulan: form.bulan ? parseInt(form.bulan) : null,
   461→   461→          status: form.status,
   462→   462→          masukBku: form.status === "lunas" ? "MASUK BKU" : null,
   463→   463→        },
   464→   464→      });
