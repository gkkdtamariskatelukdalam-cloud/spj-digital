    96→
    97→// ============ Form state ============
    98→interface FormState {
    99→  noBku: string;
   100→  uraian: string;
   101→  namaBarang: string;
   102→  vendorId: string;
   103→  vendorNewName: string;
   104→  volume: string;
   105→  satuan: string;
   106→  tarifHarga: string;
   107→  tglPesan: string;
   108→  tglBast: string;
   109→  tglBayar: string;
   110→  bulan: string;
   111→  status: "lunas" | "pending";
   112→  noPesan: string;
   113→  noBast: string;
   114→}
   115→
   116→const emptyForm: FormState = {
   117→  noBku: "",
   118→  uraian: "",
   119→  namaBarang: "",
   120→  vendorId: "__none__",
   121→  vendorNewName: "",
   122→  volume: "1",
   123→  satuan: "unit",
   124→  tarifHarga: "0",
   125→  tglPesan: "",
   126→  tglBast: "",
   127→  tglBayar: "",
   128→  bulan: String(new Date().getMonth() + 1),
   129→  status: "pending",
   130→  noPesan: "",
   131→  noBast: "",
   132→};
   133→
   134→const MONTHS = Array.from({ length: 12 }, (_, i) => i + 1);
   135→
   136→export function Transactions() {
   137→  // ============ Filter state ============
   138→  const [q, setQ] = useState("");
   139→  const [bulan, setBulan] = useState<string>("all");
   140→  const [status, setStatus] = useState<string>("all");
   141→  const [vendorId, setVendorId] = useState<string>("all");
   142→
   143→  // Debounced-ish search via state (search on type, but use latest)
   144→  const [searchInput, setSearchInput] = useState("");
   145→  useEffect(() => {
   146→    const t = setTimeout(() => setQ(searchInput), 300);
   147→    return () => clearTimeout(t);
   148→  }, [searchInput]);
   149→
   150→  // ============ Query ============
   151→  const query = useMemo(
   152→    () => ({
   153→      q: q || undefined,
   154→      bulan: bulan !== "all" ? Number(bulan) : undefined,
   155→      status: status !== "all" ? status : undefined,
   156→      vendorId: vendorId !== "all" ? vendorId : undefined,
   157→      limit: 100,
   158→    }),
   159→    [q, bulan, status, vendorId]
   160→  );
   161→
   162→  const { data, isLoading, isError, error, refetch } = useTransactions(query);
   163→  const { data: vendorsData } = useVendors();
   164→
   165→  const transactions = data?.transactions ?? [];
   166→  const totalRows = data?.total ?? 0;
   167→  const totalAmount = data?.totalAmount ?? 0;
   168→  const totalRealisasi = transactions.reduce(
   169→    (sum, t) => sum + (t.realisasi || 0),
   170→    0
   171→  );
   172→  const shown = transactions.length;
   173→
   174→  // ============ Dialog state ============
   175→  const [dialogOpen, setDialogOpen] = useState(false);
   176→  const [editingId, setEditingId] = useState<string | null>(null);
   177→  const [form, setForm] = useState<FormState>(emptyForm);
   178→
   179→  // ============ Delete dialog state ============
   180→  const [deleteTarget, setDeleteTarget] = useState<Transaction | null>(null);
   181→  const [deleteOpen, setDeleteOpen] = useState(false);
   182→
   183→  // ============ Mutations ============
   184→  const createMut = useCreateTransaction();
   185→  const updateMut = useUpdateTransaction();
   186→  const deleteMut = useDeleteTransaction();
   187→
   188→  // ============ Computed jumlah ============
   189→  const computedJumlah = useMemo(() => {
   190→    const v = parseFloat(form.volume) || 0;
   191→    const t = parseFloat(form.tarifHarga) || 0;
   192→    return v * t;
   193→  }, [form.volume, form.tarifHarga]);
   194→
   195→  // ============ Form helpers ============
   196→  function openCreate() {
   197→    setEditingId(null);
   198→    setForm({
   199→      ...emptyForm,
   200→      bulan: bulan !== "all" ? bulan : String(new Date().getMonth() + 1),
   201→    });
   202→    setDialogOpen(true);
   203→  }
   204→
   205→  function openEdit(tx: Transaction) {
   206→    setEditingId(tx.id);
   207→    setForm({
   208→      noBku: tx.noBku ?? "",
   209→      uraian: tx.uraian ?? "",
   210→      namaBarang: tx.namaBarang ?? "",
   211→      vendorId: tx.vendorId ?? "__none__",
   212→      vendorNewName: "",
   213→      volume: String(tx.volume ?? 0),
   214→      satuan: tx.satuan ?? "",
   215→      tarifHarga: String(tx.tarifHarga ?? 0),
   216→      tglPesan: toInputDate(tx.tglPesan),
   217→      tglBast: toInputDate(tx.tglBast),
   218→      tglBayar: toInputDate(tx.tglBayar),
   219→      bulan: tx.bulan ? String(tx.bulan) : String(new Date().getMonth() + 1),
   220→      status: (tx.status === "lunas" ? "lunas" : "pending"),
   221→      noPesan: tx.noPesan ?? "",
   222→      noBast: tx.noBast ?? "",
   223→    });
   224→    setDialogOpen(true);
   225→  }
   226→
   227→  function update<K extends keyof FormState>(key: K, value: FormState[K]) {
   228→    setForm((prev) => ({ ...prev, [key]: value }));
   229→  }
   230→
   231→  function handleSubmit() {
   232→    if (!form.uraian.trim()) {
   233→      toast.error("Uraian wajib diisi");
   234→      return;
   235→    }
   236→    // Build payload. Note: API derives status from masukBku field; we set
   237→    // masukBku = "MASUK BKU" when status is "lunas" so the GET filter works.
   238→    const payload: Record<string, unknown> = {
   239→      noBku: form.noBku.trim() || null,
   240→      uraian: form.uraian.trim(),
   241→      namaBarang: form.namaBarang.trim() || null,
   242→      volume: parseFloat(form.volume) || 0,
   243→      satuan: form.satuan.trim() || null,
   244→      tarifHarga: parseFloat(form.tarifHarga) || 0,
   245→      jumlah: computedJumlah,
   246→      realisasi: computedJumlah,
   247→      tglPesan: form.tglPesan || null,
   248→      tglBast: form.tglBast || null,
   249→      tglBayar: form.tglBayar || null,
   250→      noPesan: form.noPesan.trim() || null,
   251→      noBast: form.noBast.trim() || null,
   252→      bulan: Number(form.bulan),
   253→      tahun: new Date().getFullYear(),
   254→      status: form.status,
   255→      masukBku: form.status === "lunas" ? "MASUK BKU" : null,
   256→      vendorId: form.vendorId === "__none__" ? null : form.vendorId,
   257→    };
   258→
   259→    if (editingId) {
   260→      updateMut.mutate(
   261→        { id: editingId, data: payload },
   262→        {
   263→          onSuccess: () => {
   264→            toast.success("Transaksi diperbarui");
   265→            setDialogOpen(false);
   266→          },
   267→          onError: (e: Error) => toast.error("Gagal memperbarui: " + e.message),
   268→        }
   269→      );
   270→    } else {
   271→      createMut.mutate(payload, {
   272→        onSuccess: () => {
   273→          toast.success("Transaksi ditambahkan");
   274→          setDialogOpen(false);
   275→        },
   276→        onError: (e: Error) => toast.error("Gagal menambah: " + e.message),
   277→      });
   278→    }
   279→  }
   280→
   281→  function handleDelete() {
   282→    if (!deleteTarget) return;
   283→    deleteMut.mutate(deleteTarget.id, {
   284→      onSuccess: () => {
   285→        toast.success("Transaksi dihapus");
   286→        setDeleteOpen(false);
   287→        setDeleteTarget(null);
   288→      },
   289→      onError: (e: Error) => toast.error("Gagal menghapus: " + e.message),
   290→    });
   291→  }
   292→
   293→  const isSaving = createMut.isPending || updateMut.isPending;
   294→
   295→  // ============ Render ============
   296→  return (
   297→    <div className="space-y-4">
   298→      {/* === Filter bar === */}
   299→      <Card className="border-slate-200 dark:border-slate-800 bg-white/80 dark:bg-slate-950/60 backdrop-blur sticky top-0 z-30 rounded-xl shadow-sm">
   300→        <CardContent className="p-3 sm:p-4 space-y-3">
   301→          <div className="flex flex-col lg:flex-row gap-3 lg:items-end">
   302→            {/* Search */}
   303→            <div className="flex-1 min-w-0">
   304→              <Label className="text-xs text-slate-500 mb-1 block">
   305→                Cari transaksi
   306→              </Label>
   307→              <div className="relative">
   308→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-slate-400" />
   309→                <Input
   310→                  value={searchInput}
   311→                  onChange={(e) => setSearchInput(e.target.value)}
   312→                  placeholder="Uraian, nama barang, no. BKU, no. pesan..."
   313→                  className="pl-8"
   314→                />
   315→              </div>
   316→            </div>
   317→
   318→            {/* Bulan */}
   319→            <div className="w-full sm:w-36">
   320→              <Label className="text-xs text-slate-500 mb-1 block">Bulan</Label>
   321→              <Select value={bulan} onValueChange={setBulan}>
   322→                <SelectTrigger className="w-full">
   323→                  <SelectValue placeholder="Semua Bulan" />
   324→                </SelectTrigger>
   325→                <SelectContent>
   326→                  <SelectItem value="all">Semua Bulan</SelectItem>
   327→                  {MONTHS.map((m) => (
   328→                    <SelectItem key={m} value={String(m)}>
   329→                      {getMonthName(m)}
   330→                    </SelectItem>
   331→                  ))}
   332→                </SelectContent>
   333→              </Select>
   334→            </div>
   335→
   336→            {/* Status */}
   337→            <div className="w-full sm:w-36">
   338→              <Label className="text-xs text-slate-500 mb-1 block">Status</Label>
   339→              <Select value={status} onValueChange={setStatus}>
   340→                <SelectTrigger className="w-full">
   341→                  <SelectValue placeholder="Semua Status" />
   342→                </SelectTrigger>
   343→                <SelectContent>
   344→                  <SelectItem value="all">Semua Status</SelectItem>
   345→                  <SelectItem value="lunas">Lunas</SelectItem>
   346→                  <SelectItem value="pending">Pending</SelectItem>
   347→                </SelectContent>
   348→              </Select>
   349→            </div>
   350→
   351→            {/* Vendor */}
   352→            <div className="w-full sm:w-44">
   353→              <Label className="text-xs text-slate-500 mb-1 block">Vendor</Label>
   354→              <Select value={vendorId} onValueChange={setVendorId}>
   355→                <SelectTrigger className="w-full">
   356→                  <SelectValue placeholder="Semua Vendor" />
   357→                </SelectTrigger>
   358→                <SelectContent>
   359→                  <SelectItem value="all">Semua Vendor</SelectItem>
   360→                  {(vendorsData?.items ?? []).map((v) => (
   361→                    <SelectItem key={v.id} value={v.id}>
   362→                      {v.name}
   363→                    </SelectItem>
   364→                  ))}
   365→                </SelectContent>
   366→              </Select>
   367→            </div>
   368→
   369→            {/* Add button */}
   370→            <Button
   371→              onClick={openCreate}
   372→              className="bg-rose-600 hover:bg-rose-700 text-white w-full sm:w-auto"
   373→            >
   374→              <Plus className="size-4" />
   375→              Tambah Transaksi
   376→            </Button>
   377→          </div>
   378→        </CardContent>
   379→      </Card>
   380→
   381→      {/* === Summary bar === */}
   382→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   383→        <SummaryTile
   384→          icon={<ClipboardList className="size-4" />}
   385→          label="Total Transaksi"
   386→          value={formatNumber(totalRows)}
   387→          tone="violet"
   388→        />
   389→        <SummaryTile
   390→          icon={<Wallet className="size-4" />}
   391→          label="Total Nilai"
   392→          value={formatRupiah(totalAmount)}
   393→          tone="rose"
   394→        />
   395→        <SummaryTile
   396→          icon={<CheckCircle2 className="size-4" />}
   397→          label="Total Realisasi"
   398→          value={formatRupiah(totalRealisasi)}
   399→          tone="emerald"
   400→        />
   401→      </div>
   402→
   403→      {/* === Loading / Error === */}
   404→      {isLoading ? (
   405→        <Card className="border-slate-200 dark:border-slate-800">
   406→          <CardContent className="py-16 flex flex-col items-center justify-center gap-3 text-slate-500">
   407→            <Loader2 className="size-6 animate-spin text-rose-500" />
   408→            <p className="text-sm">Memuat transaksi...</p>
   409→          </CardContent>
   410→        </Card>
   411→      ) : isError ? (
   412→        <Card className="border-rose-200 dark:border-rose-900">
   413→          <CardContent className="py-12 flex flex-col items-center gap-3 text-rose-700 dark:text-rose-300">
   414→            <AlertCircle className="size-6" />
   415→            <p className="text-sm">Gagal memuat transaksi</p>
   416→            <p className="text-xs text-slate-500">
   417→              {(error as Error)?.message ?? "Unknown error"}
   418→            </p>
   419→            <Button variant="outline" onClick={() => refetch()}>
   420→              Coba lagi
   421→            </Button>
   422→          </CardContent>
   423→        </Card>
   424→      ) : shown === 0 ? (
   425→        <Card className="border-dashed border-slate-300 dark:border-slate-700">
   426→          <CardContent className="py-16 flex flex-col items-center gap-3 text-slate-500">
   427→            <Layers className="size-6 text-slate-400" />
   428→            <p className="text-sm font-medium">Belum ada transaksi</p>
   429→            <p className="text-xs">
   430→              Tambah transaksi pertama dengan tombol di atas.
   431→            </p>
   432→          </CardContent>
   433→        </Card>
   434→      ) : (
   435→        <>
   436→          {/* === Desktop table === */}
   437→          <Card className="hidden md:block border-slate-200 dark:border-slate-800 overflow-hidden">
   438→            <div className="max-h-[calc(100vh-22rem)] overflow-auto">
   439→              <Table>
   440→                <TableHeader className="sticky top-0 z-10 bg-slate-50 dark:bg-slate-900 shadow-[0_1px_0_0_rgb(0_0_0_/_0.05)]">
   441→                  <TableRow className="hover:bg-transparent">
   442→                    <TableHead className="w-10 text-center">No</TableHead>
   443→                    <TableHead className="min-w-28">No. BKU</TableHead>
   444→                    <TableHead className="min-w-32">Tgl Pesan / Bayar</TableHead>
   445→                    <TableHead className="min-w-48">Uraian</TableHead>
   446→                    <TableHead className="min-w-32">Nama Barang</TableHead>
   447→                    <TableHead className="min-w-32">Vendor</TableHead>
   448→                    <TableHead className="text-right min-w-24">Volume</TableHead>
   449→                    <TableHead className="text-right min-w-28">Tarif</TableHead>
   450→                    <TableHead className="text-right min-w-32">Jumlah</TableHead>
   451→                    <TableHead className="text-center min-w-20">Status</TableHead>
   452→                    <TableHead className="w-10 text-center">Aksi</TableHead>
   453→                  </TableRow>
   454→                </TableHeader>
   455→                <TableBody>
   456→                  {transactions.map((tx, idx) => (
   457→                    <TableRow
   458→                      key={tx.id}
   459→                      className="hover:bg-rose-50/40 dark:hover:bg-rose-950/10 transition-colors"
   460→                    >
   461→                      <TableCell className="text-center text-slate-400 text-xs">
   462→                        {idx + 1}
   463→                      </TableCell>
   464→                      <TableCell>
   465→                        {tx.noBku ? (
   466→                          <Badge
   467→                            variant="outline"
   468→                            className="font-mono text-[10px] bg-slate-50 dark:bg-slate-900 border-slate-200 dark:border-slate-700"
   469→                          >
   470→                            {tx.noBku}
   471→                          </Badge>
   472→                        ) : (
   473→                          <span className="text-slate-300">—</span>
   474→                        )}
   475→                      </TableCell>
   476→                      <TableCell className="text-xs">
   477→                        <div className="flex flex-col">
   478→                          <span>{formatDateShort(tx.tglPesan) || "—"}</span>
   479→                          {tx.tglBayar && (
   480→                            <span className="text-emerald-600 dark:text-emerald-400">
   481→                              ↳ {formatDateShort(tx.tglBayar)}
   482→                            </span>
   483→                          )}
   484→                        </div>
   485→                      </TableCell>
   486→                      <TableCell className="max-w-xs">
   487→                        <Tooltip>
   488→                          <TooltipTrigger asChild>
   489→                            <span className="line-clamp-2 cursor-help">
   490→                              {tx.uraian}
   491→                            </span>
   492→                          </TooltipTrigger>
   493→                          <TooltipContent className="max-w-sm">
   494→                            {tx.uraian}
   495→                          </TooltipContent>
   496→                        </Tooltip>
   497→                      </TableCell>
   498→                      <TableCell className="max-w-[10rem] truncate text-slate-600 dark:text-slate-300">
   499→                        {tx.namaBarang ?? "—"}
   500→                      </TableCell>
   501→                      <TableCell className="max-w-[10rem] truncate">
   502→                        {tx.vendor?.name ?? (
   503→                          <span className="text-slate-300">—</span>
   504→                        )}
   505→                      </TableCell>
   506→                      <TableCell className="text-right text-xs">
   507→                        {formatNumber(tx.volume)}{" "}
   508→                        <span className="text-slate-400">{tx.satuan}</span>
   509→                      </TableCell>
   510→                      <TableCell className="text-right text-xs text-slate-600 dark:text-slate-300">
   511→                        {formatRupiah(tx.tarifHarga)}
   512→                      </TableCell>
   513→                      <TableCell className="text-right font-bold text-rose-600 dark:text-rose-400">
   514→                        {formatRupiah(tx.jumlah)}
   515→                      </TableCell>
   516→                      <TableCell className="text-center">
   517→                        <StatusBadge status={tx.status} />
   518→                      </TableCell>
   519→                      <TableCell className="text-center">
   520→                        <RowActions
   521→                          onEdit={() => openEdit(tx)}
   522→                          onDelete={() => {
   523→                            setDeleteTarget(tx);
   524→                            setDeleteOpen(true);
   525→                          }}
   526→                        />
   527→                      </TableCell>
   528→                    </TableRow>
   529→                  ))}
   530→                </TableBody>
   531→              </Table>
   532→            </div>
   533→          </Card>
   534→
   535→          {/* === Mobile card list === */}
   536→          <div className="md:hidden space-y-3">
   537→            {transactions.map((tx, idx) => (
   538→              <TransactionCard
   539→                key={tx.id}
   540→                tx={tx}
   541→                no={idx + 1}
   542→                onEdit={() => openEdit(tx)}
   543→                onDelete={() => {
   544→                  setDeleteTarget(tx);
   545→                  setDeleteOpen(true);
   546→                }}
   547→              />
   548→            ))}
   549→          </div>
   550→
   551→          {/* === Footer note === */}
   552→          <div className="flex items-center justify-between text-xs text-slate-500 px-1">
   553→            <span>
   554→              Menampilkan <strong className="text-slate-700 dark:text-slate-300">{shown}</strong>{" "}
   555→              dari <strong className="text-slate-700 dark:text-slate-300">{formatNumber(totalRows)}</strong>{" "}
   556→              transaksi
   557→            </span>
   558→            {totalRows > 100 && (
   559→              <Badge
   560→                variant="outline"
   561→                className="text-amber-700 border-amber-300 bg-amber-50 dark:bg-amber-950/40 dark:border-amber-900 dark:text-amber-300"
   562→              >
   563→                Batas 100 baris — perbaiki filter
   564→              </Badge>
   565→            )}
   566→          </div>
   567→        </>
   568→      )}
   569→
   570→      {/* === Add/Edit Dialog === */}
   571→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   572→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   573→          <DialogHeader>
   574→            <DialogTitle className="flex items-center gap-2">

  1060→// Convert ISO date string (or other parseable formats) → yyyy-mm-dd for <input type=date>
  1061→function toInputDate(dateStr: string | null | undefined): string {
  1062→  if (!dateStr) return "";
  1063→  try {
  1064→    const d = new Date(dateStr);
  1065→    if (isNaN(d.getTime())) return "";
  1066→    const yyyy = d.getFullYear();
  1067→    const mm = String(d.getMonth() + 1).padStart(2, "0");
  1068→    const dd = String(d.getDate()).padStart(2, "0");
  1069→    return `${yyyy}-${mm}-${dd}`;
  1070→  } catch {
  1071→    return "";
  1072→  }
  1073→}
