    64→  const [search, setSearch] = useState("");
    65→  const [bulanFilter, setBulanFilter] = useState("all");

    67→  const [editingTx, setEditingTx] = useState<Transaction | null>(null);

    75→    q: search || undefined,

    96→  return (

   126→          color="text-violet-700 dark:text-violet-300"
   127→          bg="bg-violet-50 dark:bg-violet-950/40"
   128→        />
   129→        <StatCard
   130→          icon={<FileEdit className="h-4 w-4" />}
   131→          label="Draft (Belum Lengkap)"
   132→          value={stats.draft}
   133→          color="text-amber-700 dark:text-amber-300"
   134→          bg="bg-amber-50 dark:bg-amber-950/40"
   135→        />
   136→        <StatCard
   137→          icon={<Clock className="h-4 w-4" />}
   138→          label="Pending"
   139→          value={stats.pending}
   140→          color="text-cyan-700 dark:text-cyan-300"
   141→          bg="bg-cyan-50 dark:bg-cyan-950/40"
   142→        />

   144→          icon={<CheckCircle2 className="h-4 w-4" />}
   145→          label="Lunas"
   146→          value={stats.lunas}
   147→          color="text-emerald-700 dark:text-emerald-300"
   148→          bg="bg-emerald-50 dark:bg-emerald-950/40"
   149→        />

   151→
   152→      {/* Filters */}
   153→      <Card>
   154→        <CardHeader className="pb-2">
   155→          <div className="grid grid-cols-1 sm:grid-cols-4 gap-2">
   156→            <div className="relative sm:col-span-2">
   157→              <Search className="absolute left-2.5 top-2 h-4 w-4 text-muted-foreground" />

   159→                placeholder="Cari uraian, nama barang, no. BKU..."
   160→                value={search}
   161→                onChange={(e) => setSearch(e.target.value)}
   162→                className="pl-9 h-9 text-sm"

   177→            </Select>
   178→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   179→              <SelectTrigger className="h-9 text-sm">
   180→                <SelectValue placeholder="Semua Status" />

   196→                <SelectItem value="lunas">
   197→                  <span className="flex items-center gap-1.5">
   198→                    <CheckCircle2 className="h-3 w-3 text-emerald-600" />

   211→                Memuat data belanja...
   212→              </span>
   213→            </div>
   214→          ) : error ? (
   215→            <div className="p-6 text-center text-sm text-rose-600">
   216→              <AlertTriangle className="h-5 w-5 mx-auto mb-2" />
   217→              Gagal memuat data: {error.message}
   218→            </div>

   220→            <div className="p-12 text-center">
   221→              <ShoppingCart className="h-10 w-10 text-muted-foreground/40 mx-auto mb-3" />
   222→              <p className="text-sm font-medium">Belum ada data belanja</p>
   223→              <p className="text-xs text-muted-foreground mt-1">
   224→                Import Excel atau tambah manual untuk mulai

   228→            <ScrollArea className="h-[600px]">
   229→              <Table>
   230→                <TableHeader className="sticky top-0 bg-card z-10">

   254→              </Table>

   256→          )}
   257→        </CardContent>
   258→      </Card>
   259→
   260→      {transactions.length > 0 && (
   261→        <div className="text-xs text-muted-foreground text-center">
   262→          Menampilkan {transactions.length} data belanja · Total Nilai:{" "}
   263→          <span className="font-mono font-bold text-rose-700 dark:text-rose-300">
   264→            {formatRupiah(stats.totalNilai)}
   265→          </span>
   266→        </div>
   267→      )}
   268→
   269→      {/* Edit Dialog */}
   270→      {editingTx && (
   271→        <EditBelanjaDialog
   272→          tx={editingTx}
   273→          onClose={() => setEditingTx(null)}
   274→        />
   275→      )}
   276→
   277→      {/* Add Dialog */}
   278→      {showAdd && <AddBelanjaDialog onClose={() => setShowAdd(false)} />}
   279→    </div>
   280→  );
   281→}
   282→
   283→function BelanjaRow({
   284→  tx,
   285→  index,
   286→  onEdit,
   287→}: {
   288→  tx: Transaction;
   289→  index: number;
   290→  onEdit: () => void;
   291→}) {
   292→  const isDraft = tx.status === "draft";
   293→  const isLunas = tx.status === "lunas";
   294→
   295→  // Check what's missing
   296→  const missing = [];
   297→  if (!tx.uraian) missing.push("uraian");
   298→  if (!tx.namaBarang) missing.push("namaBarang");
   299→  if (!tx.volume) missing.push("volume");
   300→  if (!tx.tarifHarga) missing.push("harga");
   301→  if (!tx.tglPesan) missing.push("tglPesan");
   302→  if (!tx.vendorId) missing.push("vendor");
   303→
   304→  return (
   305→    <TableRow
   306→      className={cn(
   307→        "hover:bg-muted/40",
   308→        isDraft && "bg-amber-50/40 dark:bg-amber-950/10"
   309→      )}
   310→    >
   311→      <TableCell className="text-xs font-mono text-muted-foreground">
   312→        {index}
   313→      </TableCell>
   314→      <TableCell className="text-xs font-mono">
   315→        {tx.noBku || <span className="text-muted-foreground/40">—</span>}
   316→      </TableCell>
   317→      <TableCell className="text-xs">
   318→        {tx.tglPesan ? (
   319→          formatDate(tx.tglPesan)
   320→        ) : (
   321→          <span className="text-muted-foreground/40">—</span>
   322→        )}
   323→      </TableCell>
   324→      <TableCell className="text-xs">
   325→        <div className="font-medium truncate max-w-[200px]" title={tx.uraian}>
   326→          {tx.uraian || <span className="text-muted-foreground/40 italic">Belum diisi</span>}
   327→        </div>
   328→        {tx.namaBarang && (
   329→          <div className="text-[10px] text-muted-foreground truncate max-w-[200px]">
   330→            {tx.namaBarang}
   331→          </div>
   332→        )}
   333→      </TableCell>
   334→      <TableCell className="text-xs font-mono text-center">
   335→        {tx.volume > 0 ? (
   336→          `${tx.volume} ${tx.satuan || ""}`
   337→        ) : (
   338→          <span className="text-muted-foreground/40">—</span>
   339→        )}
   340→      </TableCell>
   341→      <TableCell className="text-xs font-mono text-right">
   342→        {tx.tarifHarga > 0 ? (
   343→          formatRupiah(tx.tarifHarga)
   344→        ) : (
   345→          <span className="text-muted-foreground/40">—</span>
   346→        )}
   347→      </TableCell>
   348→      <TableCell className="text-xs font-mono text-right font-semibold">
   349→        {tx.jumlah > 0 ? (
   350→          <span className="text-rose-700 dark:text-rose-300">
   351→            {formatRupiah(tx.jumlah)}
   352→          </span>
   353→        ) : (
   354→          <span className="text-muted-foreground/40">—</span>
   355→        )}
   356→      </TableCell>
   357→      <TableCell className="text-xs">
   358→        {tx.vendor?.name || (
   359→          <span className="text-muted-foreground/40">—</span>
   360→        )}
   361→      </TableCell>
   362→      <TableCell>
   363→        {isDraft ? (
   364→          <Badge
   365→            variant="outline"
   366→            className="text-[9px] border-amber-400 text-amber-700 dark:text-amber-300"
   367→            title={`Belum lengkap: ${missing.join(", ")}`}
   368→          >
   369→            <FileEdit className="h-2.5 w-2.5 mr-0.5" />
   370→            Draft
   371→          </Badge>
   372→        ) : isLunas ? (
   373→          <Badge
   374→            variant="outline"
   375→            className="text-[9px] border-emerald-400 text-emerald-700 dark:text-emerald-300"
   376→          >
   377→            <CheckCircle2 className="h-2.5 w-2.5 mr-0.5" />
   378→            Lunas
   379→          </Badge>
   380→        ) : (
   381→          <Badge
   382→            variant="outline"
   383→            className="text-[9px] border-cyan-400 text-cyan-700 dark:text-cyan-300"
   384→          >
   385→            <Clock className="h-2.5 w-2.5 mr-0.5" />
   386→            Pending
   387→          </Badge>
   388→        )}
   389→      </TableCell>
   390→      <TableCell>
   391→        <Button
   392→          variant="ghost"
   393→          size="sm"
   394→          onClick={onEdit}
   395→          className="h-7 w-7 p-0"
   396→          title={isDraft ? "Lengkapi data" : "Edit data"}
   397→        >
   398→          <Pencil className="h-3 w-3" />
   399→        </Button>
   400→      </TableCell>
   401→    </TableRow>
   402→  );
   403→}

   405→function EditBelanjaDialog({
   406→  tx,
   407→  onClose,
   408→}: {
   409→  tx: Transaction;
   410→  onClose: () => void;
   411→}) {
   412→  const updateMutation = useUpdateTransaction();
   413→  const deleteMutation = useDeleteTransaction();
   414→  const { data: vendorsData } = useVendors();
   415→  const vendors = vendorsData?.items ?? [];
   416→
   417→  const [form, setForm] = useState({
   418→    noBku: tx.noBku || "",
   419→    noPesan: tx.noPesan || "",
   420→    tglPesan: tx.tglPesan && tx.tglPesan.includes("-")
   421→      ? tx.tglPesan.substring(0, 10)
   422→      : tx.tglPesan || "",
   423→    tglBast: tx.tglBast && tx.tglBast.includes("-")
   424→      ? tx.tglBast.substring(0, 10)
   425→      : tx.tglBast || "",
   426→    tglBayar: tx.tglBayar && tx.tglBayar.includes("-")
   427→      ? tx.tglBayar.substring(0, 10)
   428→      : tx.tglBayar || "",
   429→    uraian: tx.uraian || "",
   430→    namaBarang: tx.namaBarang || "",
   431→    volume: String(tx.volume || 0),
   432→    satuan: tx.satuan || "",
   433→    tarifHarga: String(tx.tarifHarga || 0),
   434→    vendorId: tx.vendorId || "__none__",
   435→    bulan: String(tx.bulan || ""),
   436→    status: tx.status || "pending",
   437→  });
   438→
   439→  const computedJumlah =
   440→    (parseFloat(form.volume) || 0) * (parseFloat(form.tarifHarga) || 0);
   441→
   442→  const handleSave = async () => {
   443→    try {
   444→      await updateMutation.mutateAsync({
   445→        id: tx.id,
   446→        data: {
   447→          noBku: form.noBku || null,
   448→          noPesan: form.noPesan || null,
   449→          tglPesan: form.tglPesan || null,
   450→          tglBast: form.tglBast || null,
   451→          tglBayar: form.tglBayar || null,
   452→          uraian: form.uraian,
   453→          namaBarang: form.namaBarang || null,
   454→          volume: parseFloat(form.volume) || 0,
   455→          satuan: form.satuan || null,
   456→          tarifHarga: parseFloat(form.tarifHarga) || 0,
   457→          jumlah: computedJumlah,
   458→          realisasi: computedJumlah,
   459→          vendorId: form.vendorId && form.vendorId !== "__none__" ? form.vendorId : null,
   460→          bulan: form.bulan ? parseInt(form.bulan) : null,
   461→          status: form.status,
   462→          masukBku: form.status === "lunas" ? "MASUK BKU" : null,
   463→        },
   464→      });
   465→      toast.success("Data belanja berhasil diperbarui");
   466→      onClose();
   467→    } catch (e) {
   468→      toast.error("Gagal menyimpan: " + (e as Error).message);
   469→    }
   470→  };
   471→
   472→  const handleDelete = async () => {
   473→    if (!confirm("Hapus data belanja ini?")) return;
   474→    try {
   475→      await deleteMutation.mutateAsync(tx.id);
   476→      toast.success("Data berhasil dihapus");
   477→      onClose();
   478→    } catch (e) {
   479→      toast.error("Gagal menghapus: " + (e as Error).message);
   480→    }
   481→  };
   482→
   483→  return (
   484→    <Dialog open onOpenChange={onClose}>
   485→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   486→        <DialogHeader>
   487→          <DialogTitle className="flex items-center gap-2">
   488→            <Pencil className="h-4 w-4 text-violet-600" />
   489→            Edit Data Belanja
   490→            {tx.status === "draft" && (
   491→              <Badge variant="outline" className="text-[10px] border-amber-400 text-amber-700 dark:text-amber-300">
   492→                Draft - Lengkapi data
   493→              </Badge>
   494→            )}
   495→          </DialogTitle>
   496→        </DialogHeader>
   497→
   498→        <div className="grid grid-cols-2 gap-3 py-2">
   499→          <div>
   500→            <Label className="text-xs">No. BKU</Label>
   501→            <Input
   502→              value={form.noBku}
   503→              onChange={(e) => setForm({ ...form, noBku: e.target.value })}
   504→              placeholder="BPU01"
   505→              className="h-9 text-sm font-mono"
   506→            />
   507→          </div>
   508→          <div>
   509→            <Label className="text-xs">No. Pesan</Label>
   510→            <Input
   511→              value={form.noPesan}
   512→              onChange={(e) => setForm({ ...form, noPesan: e.target.value })}
   513→              placeholder="01"
   514→              className="h-9 text-sm font-mono"
   515→            />
   516→          </div>
   517→          <div className="col-span-2">
   518→            <Label className="text-xs">Uraian Kegiatan *</Label>
   519→            <Textarea
   520→              value={form.uraian}
   521→              onChange={(e) => setForm({ ...form, uraian: e.target.value })}
   522→              placeholder="cth. Pengadaan ATK untuk kegiatan KBM"
   523→              className="text-sm min-h-[60px]"
   524→            />
   525→          </div>
   526→          <div className="col-span-2">
   527→            <Label className="text-xs">Nama Barang</Label>
   528→            <Input
   529→              value={form.namaBarang}
   530→              onChange={(e) => setForm({ ...form, namaBarang: e.target.value })}
   531→              placeholder="cth. Kertas HVS A4 80gsm"
   532→              className="h-9 text-sm"
   533→            />
   534→          </div>
   535→          <div>
   536→            <Label className="text-xs">Volume</Label>
   537→            <Input
   538→              type="number"
   539→              value={form.volume}
   540→              onChange={(e) => setForm({ ...form, volume: e.target.value })}
   541→              className="h-9 text-sm font-mono"
   542→            />
   543→          </div>
   544→          <div>
   545→            <Label className="text-xs">Satuan</Label>
   546→            <Input
   547→              value={form.satuan}
   548→              onChange={(e) => setForm({ ...form, satuan: e.target.value })}
   549→              placeholder="rim, dus, buah"
   550→              className="h-9 text-sm"
   551→            />
   552→          </div>
   553→          <div>
   554→            <Label className="text-xs">Harga Satuan (Rp)</Label>
   555→            <Input
   556→              type="number"
   557→              value={form.tarifHarga}
   558→              onChange={(e) => setForm({ ...form, tarifHarga: e.target.value })}
   559→              className="h-9 text-sm font-mono"
