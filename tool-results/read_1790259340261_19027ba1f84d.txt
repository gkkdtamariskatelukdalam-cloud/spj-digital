    93→    if (selectedFile) {
    94→      handleFile(selectedFile);
    95→    }
    96→  };
    97→
    98→  const handleDrop = (e: React.DragEvent) => {
    99→    e.preventDefault();

   101→    if (droppedFile) {
   102→      handleFile(droppedFile);
   103→    }
   104→  };
   105→
   106→  const handleDragOver = (e: React.DragEvent) => {
   107→    e.preventDefault();
   108→  };
   109→

   111→    const name = selectedFile.name.toLowerCase();

   127→    setError(null);
   128→    try {
   129→      const formData = new FormData();
   130→      formData.append("file", selectedFile);
   131→      const res = await fetch("/api/spj/import/preview", {
   132→        method: "POST",
   133→        body: formData,
   134→      });
   135→      const data = await res.json();
   136→      if (!res.ok) {
   137→        throw new Error(data.error || "Gagal parse Excel");
   138→      }
   139→      setPreview(data as PreviewResult);
   140→      toast.success(`File berhasil di-parse: ${data.totalRows} baris data`);
   141→    } catch (e) {
   142→      setError((e as Error).message);
   143→      toast.error("Gagal parse Excel: " + (e as Error).message);
   144→    } finally {
   145→      setPreviewLoading(false);
   146→    }
   147→  };

   154→      !confirm(
   155→        `Import ${preview?.totalRows || 0} baris data dari Excel? Data akan ditambahkan ke database.`
   156→      )
   157→    )
   158→      return;
   159→
   160→    setImportLoading(true);
   161→    setError(null);
   162→    try {
   163→      const formData = new FormData();
   164→      formData.append("file", file);
   165→      const res = await fetch("/api/spj/import", {
   166→        method: "POST",
   167→        body: formData,
   168→      });
   169→      const data = await res.json();
   170→      if (!res.ok) {
   171→        throw new Error(data.error || "Gagal import Excel");
   172→      }
   173→      setImportResult(data as ImportResult);
   174→      toast.success(
   175→        `Berhasil import ${data.summary.transactionsImported} transaksi!`,
   176→        { duration: 4000 }
   177→      );

   179→      setError((e as Error).message);
   180→      toast.error("Gagal import: " + (e as Error).message);

   182→      setImportLoading(false);
   183→    }
   184→  };

   186→  const handleReset = () => {
   187→    setFile(null);
   188→    setPreview(null);

   374→

   376→      <Card>

   385→
   386→          {!file ? (
   387→            <div
   388→              onDrop={handleDrop}

   390→              onClick={() => fileInputRef.current?.click()}
   391→              className="border-2 border-dashed border-slate-300 dark:border-slate-700 rounded-lg p-12 text-center cursor-pointer hover:border-emerald-400 hover:bg-emerald-50/30 dark:hover:bg-emerald-950/10 transition-all"

   398→                Format: .xlsx atau .xls (Max 10MB)
