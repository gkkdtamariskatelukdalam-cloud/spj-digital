    65→    65→const ROMAN: Record<number, string> = {
    66→    66→  1: "I",
    67→    67→  2: "II",
    68→    68→  3: "III",
    69→    69→  4: "IV",

    71→    71→  6: "VI",

    74→    74→  9: "IX",

    78→    78→};
    79→    79→
    80→    80→function toRoman(n: number | null | undefined): string {
    81→    81→  if (!n || n < 1 || n > 12) return "I";
    82→    82→  return ROMAN[n] ?? "I";
    83→    83→}
    84→    84→
    85→    85→// Build the SPJ document number from a transaction
    86→    86→function buildSpjNumber(tx: Transaction): string {
    87→    87→  const bku = tx.noBku?.trim() || String(tx.noUrut ?? tx.id.slice(-6));
    88→    88→  const roman = toRoman(tx.bulan);
    89→    89→  return `SPJ-${bku}/${roman}/${tx.tahun}`;
    90→    90→}
    91→    91→
    92→    92→// Pick the most relevant date for the SPJ doc (prefer payment, fallback to BAST/Pesan/today)
    93→    93→function pickSpjDate(tx: Transaction): string {
    94→    94→  return tx.tglBayar || tx.tglBast || tx.tglPesan || new Date().toISOString();
    95→    95→}
    96→    96→
    97→    97→// Capitalize first letter (terbilang returns lowercase)
    98→    98→function capitalize(s: string): string {
    99→    99→  if (!s) return s;
   100→   100→  return s.charAt(0).toUpperCase() + s.slice(1);
   101→   101→}
