    63→const MONTHS = Array.from({ length: 12 }, (_, i) => i + 1);
    64→
    65→const ROMAN: Record<number, string> = {
    66→  1: "I",
    67→  2: "II",
    68→  3: "III",
    69→  4: "IV",
    70→  5: "V",
    71→  6: "VI",
    72→  7: "VII",
    73→  8: "VIII",
    74→  9: "IX",
    75→  10: "X",
    76→  11: "XI",
    77→  12: "XII",
    78→};
    79→
    80→function toRoman(n: number | null | undefined): string {
    81→  if (!n || n < 1 || n > 12) return "I";
    82→  return ROMAN[n] ?? "I";
    83→}
    84→
    85→// Build the SPJ document number from a transaction
    86→function buildSpjNumber(tx: Transaction): string {
    87→  const bku = tx.noBku?.trim() || String(tx.noUrut ?? tx.id.slice(-6));
    88→  const roman = toRoman(tx.bulan);
    89→  return `SPJ-${bku}/${roman}/${tx.tahun}`;
    90→}
    91→
    92→// Pick the most relevant date for the SPJ doc (prefer payment, fallback to BAST/Pesan/today)
    93→function pickSpjDate(tx: Transaction): string {
    94→  return tx.tglBayar || tx.tglBast || tx.tglPesan || new Date().toISOString();
    95→}
    96→
    97→// Capitalize first letter (terbilang returns lowercase)
    98→function capitalize(s: string): string {
    99→  if (!s) return s;
   100→  return s.charAt(0).toUpperCase() + s.slice(1);
   101→}
