    67→// CmInput: Uncontrolled input for cm values.
    68→// User can type freely (e.g., "3", "2.91") without value reverting.
    69→// key = rounded cm value → only re-mounts when cm changes by ≥1 (settings load)
    70→// onBlur → convert cm to px, call parent onChange
    71→function CmInput({ pxValue, onChange, label }: { pxValue: number; onChange: (px: number) => void; label: string }) {
    72→  return (
    73→    <div>
    74→      <Label className="text-[10px] text-muted-foreground">{label}</Label>
    75→      <Input
    76→        key={Math.round(pxValue / 37.795)}
    77→        type="number"
    78→        step="0.01"
    79→        min="0.5"
    80→        max="10"
    81→        defaultValue={(pxValue / 37.795).toFixed(2)}
    82→        onBlur={(e) => {
    83→          const px = Math.round(parseFloat(e.target.value || "0") * 37.795);
    84→          if (!isNaN(px) && px >= 0) onChange(px);
    85→        }}
    86→        className="h-8 text-xs font-mono"
    87→      />
    88→    </div>
    89→  );
    90→}

   835→function AppLogoUpload() {
   836→  const [appLogo, setAppLogo] = useState<string | null>(null);
   837→  const [loading, setLoading] = useState(false);
   838→
   839→  useEffect(() => {
   840→    fetch("/api/app-settings")
   841→      .then((r) => r.json())
   842→      .then((data) => setAppLogo(data.appLogo || null))
   843→      .catch(() => {});
   844→  }, []);
   845→
   846→  async function handleUpload(e: React.ChangeEvent<HTMLInputElement>) {
   847→    const file = e.target.files?.[0];
   848→    if (!file) return;
   849→    if (file.size > 5 * 1024 * 1024) {
   850→      toast.error("File terlalu besar. Maksimal 5MB.");
   851→      return;
   852→    }
   853→    setLoading(true);
   854→    const formData = new FormData();
   855→    formData.append("logo", file);
   856→    try {
   857→      const res = await fetch("/api/app-settings/upload-logo", {
   858→        method: "POST",
   859→        body: formData,
   860→      });
   861→      const data = await res.json();
   862→      if (!res.ok) throw new Error(data.error || "Gagal upload");
   863→      setAppLogo(data.appLogo);
   864→      // Sync localStorage so login + dashboard headers update on next page load
   865→      try {
   866→        localStorage.setItem("appLogo", data.appLogo);
   867→      } catch {
   868→        // localStorage might be full — ignore
   869→      }
   870→      toast.success("Logo aplikasi diperbarui.");
   871→    } catch (err) {
   872→      toast.error((err as Error).message);
   873→    } finally {
   874→      setLoading(false);
   875→    }
   876→  }
   877→
   878→  async function handleDelete() {
   879→    setLoading(true);
   880→    try {
   881→      await fetch("/api/app-settings", {
   882→        method: "PUT",
   883→        headers: { "Content-Type": "application/json" },
   884→        body: JSON.stringify({ appLogo: null }),
   885→      });
   886→      setAppLogo(null);
   887→      // Clear localStorage cache so login + dashboard headers fall back
   888→      try {
   889→        localStorage.removeItem("appLogo");
   890→      } catch {
   891→        // ignore
   892→      }
   893→      toast.success("Logo aplikasi dihapus.");
   894→    } finally {
   895→      setLoading(false);
   896→    }
   897→  }
   898→
   899→  return (
   900→    <Card className="border-l-4 border-l-violet-500">
   901→      <CardHeader className="pb-3">
   902→        <CardTitle className="text-base flex items-center gap-2">
   903→          <ImageIcon className="h-4 w-4 text-violet-600" />
   904→          Logo Aplikasi (Login + Favicon)
   905→        </CardTitle>
   906→        <CardDescription className="text-xs">
   907→          Logo ini muncul di halaman login dan sebagai favicon browser. Maksimal 5MB.
   908→        </CardDescription>
   909→      </CardHeader>
   910→      <CardContent className="space-y-3">
   911→        <div className="flex items-center gap-4">
   912→          <div className="h-16 w-16 rounded-xl bg-slate-100 dark:bg-slate-800 border-2 border-dashed border-slate-300 dark:border-slate-700 flex items-center justify-center overflow-hidden flex-shrink-0">
   913→            {appLogo ? (
   914→              <img src={appLogo} alt="App Logo" className="h-full w-full object-contain" />
   915→            ) : (
   916→              <ImageIcon className="h-6 w-6 text-slate-400" />
   917→            )}
   918→          </div>
   919→          <div className="flex flex-col gap-2">
   920→            <label className="cursor-pointer">
   921→              <span className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-md bg-violet-600 text-white text-xs font-medium hover:bg-violet-700 transition-colors">
   922→                <Upload className="h-3.5 w-3.5" />
   923→                {loading ? "Mengupload..." : "Upload Logo"}
   924→              </span>
   925→              <input type="file" accept="image/*" className="hidden" onChange={handleUpload} disabled={loading} />
   926→            </label>
   927→            {appLogo && (
   928→              <button
   929→                onClick={handleDelete}
   930→                disabled={loading}
   931→                className="text-xs text-rose-600 hover:text-rose-700"
   932→              >
   933→                Hapus Logo
   934→              </button>
   935→            )}
   936→          </div>
   937→        </div>
   938→      </CardContent>
   939→    </Card>
   940→  );
   941→}
