   137→   137→  // ============ Filter state ============
   138→   138→  const [q, setQ] = useState("");

   140→   140→  const [status, setStatus] = useState<string>("all");

   142→   142→
   143→   143→  // Debounced-ish search via state (search on type, but use latest)
   144→   144→  const [searchInput, setSearchInput] = useState("");
   145→   145→  useEffect(() => {

   148→   148→  }, [searchInput]);
   149→   149→
   150→   150→  // ============ Query ============
   151→   151→  const query = useMemo(
   152→   152→    () => ({
   153→   153→      q: q || undefined,
   154→   154→      bulan: bulan !== "all" ? Number(bulan) : undefined,
   155→   155→      status: status !== "all" ? status : undefined,
   156→   156→      vendorId: vendorId !== "all" ? vendorId : undefined,
   157→   157→      limit: 100,
   158→   158→    }),
   159→   159→    [q, bulan, status, vendorId]
   160→   160→  );
   161→   161→
   162→   162→  const { data, isLoading, isError, error, refetch } = useTransactions(query);
   163→   163→  const { data: vendorsData } = useVendors();
   164→   164→
   165→   165→  const transactions = data?.transactions ?? [];
   166→   166→  const totalRows = data?.total ?? 0;
   167→   167→  const totalAmount = data?.totalAmount ?? 0;
   168→   168→  const totalRealisasi = transactions.reduce(
   169→   169→    (sum, t) => sum + (t.realisasi || 0),
   170→   170→    0
   171→   171→  );
   172→   172→  const shown = transactions.length;
   173→   173→
   174→   174→  // ============ Dialog state ============
   175→   175→  const [dialogOpen, setDialogOpen] = useState(false);
   176→   176→  const [editingId, setEditingId] = useState<string | null>(null);
   177→   177→  const [form, setForm] = useState<FormState>(emptyForm);
   178→   178→

   193→   193→  }, [form.volume, form.tarifHarga]);
   194→   194→

   201→   201→    });
   202→   202→    setDialogOpen(true);
   203→   203→  }
   204→   204→
   205→   205→  function openEdit(tx: Transaction) {
   206→   206→    setEditingId(tx.id);
   207→   207→    setForm({
   208→   208→      noBku: tx.noBku ?? "",
   209→   209→      uraian: tx.uraian ?? "",
   210→   210→      namaBarang: tx.namaBarang ?? "",
   211→   211→      vendorId: tx.vendorId ?? "__none__",
   212→   212→      vendorNewName: "",
   213→   213→      volume: String(tx.volume ?? 0),
   214→   214→      satuan: tx.satuan ?? "",
   215→   215→      tarifHarga: String(tx.tarifHarga ?? 0),

   240→   240→      uraian: form.uraian.trim(),
   241→   241→      namaBarang: form.namaBarang.trim() || null,

   245→   245→      jumlah: computedJumlah,
   246→   246→      realisasi: computedJumlah,
   247→   247→      tglPesan: form.tglPesan || null,

   249→   249→      tglBayar: form.tglBayar || null,

   259→   259→    if (editingId) {
   260→   260→      updateMut.mutate(
   261→   261→        { id: editingId, data: payload },
   262→   262→        {
   263→   263→          onSuccess: () => {
   264→   264→            toast.success("Transaksi diperbarui");
   265→   265→            setDialogOpen(false);
   266→   266→          },
   267→   267→          onError: (e: Error) => toast.error("Gagal memperbarui: " + e.message),
   268→   268→        }
   269→   269→      );
   270→   270→    } else {
   271→   271→      createMut.mutate(payload, {
   272→   272→        onSuccess: () => {
   273→   273→          toast.success("Transaksi ditambahkan");
   274→   274→          setDialogOpen(false);
   275→   275→        },
   276→   276→        onError: (e: Error) => toast.error("Gagal menambah: " + e.message),
   277→   277→      });
   278→   278→    }
   279→   279→  }
   280→   280→
   281→   281→  function handleDelete() {

   283→   283→    deleteMut.mutate(deleteTarget.id, {
   284→   284→      onSuccess: () => {
   285→   285→        toast.success("Transaksi dihapus");
   286→   286→        setDeleteOpen(false);
   287→   287→        setDeleteTarget(null);
   288→   288→      },
   289→   289→      onError: (e: Error) => toast.error("Gagal menghapus: " + e.message),
   290→   290→    });
   291→   291→  }
   292→   292→
   293→   293→  const isSaving = createMut.isPending || updateMut.isPending;
   294→   294→
   295→   295→  // ============ Render ============
   296→   296→  return (
   297→   297→    <div className="space-y-4">
   298→   298→      {/* === Filter bar === */}

   300→   300→        <CardContent className="p-3 sm:p-4 space-y-3">
   301→   301→          <div className="flex flex-col lg:flex-row gap-3 lg:items-end">
   302→   302→            {/* Search */}
   303→   303→            <div className="flex-1 min-w-0">
   304→   304→              <Label className="text-xs text-slate-500 mb-1 block">
   305→   305→                Cari transaksi
   306→   306→              </Label>
   307→   307→              <div className="relative">
   308→   308→                <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-slate-400" />
   309→   309→                <Input
   310→   310→                  value={searchInput}

   329→   329→                      {getMonthName(m)}

   331→   331→                  ))}
   332→   332→                </SelectContent>
   333→   333→              </Select>
   334→   334→            </div>

   336→   336→            {/* Status */}

   351→   351→            {/* Vendor */}

   394→   394→        />

   397→   397→          label="Total Realisasi"
   398→   398→          value={formatRupiah(totalRealisasi)}
   399→   399→          tone="emerald"
   400→   400→        />
   401→   401→      </div>
   402→   402→
   403→   403→      {/* === Loading / Error === */}
   404→   404→      {isLoading ? (
   405→   405→        <Card className="border-slate-200 dark:border-slate-800">
   406→   406→          <CardContent className="py-16 flex flex-col items-center justify-center gap-3 text-slate-500">

   420→   420→              Coba lagi
   421→   421→            </Button>
   422→   422→          </CardContent>
   423→   423→        </Card>
   424→   424→      ) : shown === 0 ? (
   425→   425→        <Card className="border-dashed border-slate-300 dark:border-slate-700">
   426→   426→          <CardContent className="py-16 flex flex-col items-center gap-3 text-slate-500">

   431→   431→            </p>
   432→   432→          </CardContent>
   433→   433→        </Card>
   434→   434→      ) : (
   435→   435→        <>
   436→   436→          {/* === Desktop table === */}
   437→   437→          <Card className="hidden md:block border-slate-200 dark:border-slate-800 overflow-hidden">
   438→   438→            <div className="max-h-[calc(100vh-22rem)] overflow-auto">
   439→   439→              <Table>
   440→   440→                <TableHeader className="sticky top-0 z-10 bg-slate-50 dark:bg-slate-900 shadow-[0_1px_0_0_rgb(0_0_0_/_0.05)]">

   464→   464→                      <TableCell>
   465→   465→                        {tx.noBku ? (
   466→   466→                          <Badge
   467→   467→                            variant="outline"
   468→   468→                            className="font-mono text-[10px] bg-slate-50 dark:bg-slate-900 border-slate-200 dark:border-slate-700"
   469→   469→                          >
   470→   470→                            {tx.noBku}
   471→   471→                          </Badge>
   472→   472→                        ) : (
   473→   473→                          <span className="text-slate-300">—</span>
