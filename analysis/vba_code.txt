Private Sub Worksheet_Change(ByVal Target As Range)
    On Error GoTo ErrHandler
    Dim wsBantu As Worksheet
    Set wsBantu = ThisWorkbook.Sheets("Tabel Bantu")
    Dim cell As Range
    
    Application.EnableEvents = False
    For Each cell In Target
        If cell.Value <> wsBantu.Cells(cell.Row, cell.Column).Value Then
            If Len(cell.Value) = 0 Then ' Cell di "Master" terhapus
                wsBantu.Cells(cell.Row, cell.Column).Interior.Color = RGB(255, 0, 0) ' Merah
            Else
                wsBantu.Cells(cell.Row, cell.Column).Interior.Color = RGB(255, 255, 0) ' Kuning
            End If
        Else
            wsBantu.Cells(cell.Row, cell.Column).Interior.ColorIndex = 0 ' Kembali ke warna semula
        End If
    Next cell
    Application.EnableEvents = True
    Exit Sub
    
ErrHandler:
    MsgBox "Error: " & Err.Description
    Application.EnableEvents = True
End Sub

Private Sub Worksheet_Change(ByVal Target As Range)
    ' Periksa apakah perubahan terjadi pada cell target di sheet "04SHP"
    If Me.Name = "04SHP" And Not Intersect(Target, Me.Range("P4")) Is Nothing Then
        ' Panggil macro HideEmptyRowsBasedOnColumnB
        Call HideEmptyRowsBasedOnColumnB
    End If
End Sub

Sub HideEmptyRowsBasedOnColumnB()
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim k As Long
    
    ' Set worksheet "04SHP" sebagai worksheet target
    Set ws = ThisWorkbook.Sheets("04SHP")
    
    ' Hanya proses lembar kerja yang tidak dilindungi
    If Not ws.ProtectContents Then
        ' Tentukan baris terakhir dengan data di kolom B
        lastRow = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row
        
        ' Batasan untuk loop dari baris 24 hingga 93
        If lastRow > 93 Then lastRow = 93
        
        ' Loop melalui setiap baris dari baris 24 hingga lastRow
        For k = 24 To 93
            ' Cek jika kolom B pada baris tersebut tidak kosong
            If Len(ws.Range("B" & k).Value) > 0 Then
                ' Jika tidak kosong, tampilkan baris
                ws.Rows(k).Hidden = False
            Else
                ' Jika kosong, sembunyikan baris
                ws.Rows(k).Hidden = True
            End If
        Next k
    End If
End Sub

Sub SavePDF()
    Dim ws As Worksheet
    Dim fileName As String
    
    ' Tentukan worksheet "01PESAN" sebagai worksheet target
    Set ws = ThisWorkbook.Sheets("01PESAN")
    
    ' Periksa nilai di cell P4 untuk mendapatkan bagian tengah nama file
    Dim middleName As String
    middleName = ws.Range("P4").Value
    
    ' Periksa apakah cell P4 tidak kosong
    If Len(middleName) > 0 Then
        ' Bangun nama file PDF
        fileName = "01PESAN_" & middleName & "_2025.pdf"
        
        ' Simpan file PDF
        ws.ExportAsFixedFormat Type:=xlTypePDF, fileName:=ThisWorkbook.Path & "\" & fileName, Quality:=xlQualityStandard, IncludeDocProperties:=True, IgnorePrintAreas:=False, OpenAfterPublish:=False
        
        ' Beri pesan berhasil
        MsgBox "File PDF berhasil disimpan dengan nama: " & fileName, vbInformation
    Else
        ' Beri pesan gagal karena cell P4 kosong
        MsgBox "Gagal menyimpan file PDF karena cell P4 kosong.", vbExclamation
    End If
End Sub
Private Sub Worksheet_Change(ByVal Target As Range)
    ' Periksa apakah perubahan terjadi pada cell P4
    If Not Intersect(Target, Me.Range("P4")) Is Nothing Then
        ' Panggil macro SavePDF
        Call SavePDF
    End If
End Sub

        ' Beri pesan gagal karena cell P4 kosong
        MsgBox "Gagal menyimpan file PDF karena cell P4 kosong.", vbExclamation
    End If
End Sub
Private Sub Worksheet_Change(ByVal Target As Range)
    ' Periksa apakah perubahan terjadi pada cell P4
    If Not Intersect(Target, Me.Range("P4")) Is Nothing Then
        ' Panggil macro SavePDF
        Call SavePDF
    End If
End Sub
